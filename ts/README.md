# @digitallink-global/api-sdk@0.1.0

A TypeScript SDK client for the app.dgnlinks.com API.

## Usage

First, install the SDK from npm.

```bash
npm install @digitallink-global/api-sdk --save
```

Next, try it out.


```ts
import {
  Configuration,
  AccountApi,
} from '@digitallink-global/api-sdk';
import type { GetAccountRequest } from '@digitallink-global/api-sdk';

async function example() {
  console.log("🚀 Testing @digitallink-global/api-sdk SDK...");
  const config = new Configuration({ 
    // Configure HTTP bearer authorization: apiKey
    accessToken: "YOUR BEARER TOKEN",
  });
  const api = new AccountApi(config);

  try {
    const data = await api.getAccount();
    console.log(data);
  } catch (error) {
    console.error(error);
  }
}

// Run the test
example().catch(console.error);
```
